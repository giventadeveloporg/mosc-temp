Defines all DDL for the Kids Festival feature. Follow .cursor/rules/database_schema_guidelines.mdc: bigint PKs via sequence_generator, tenant_id on tenant-scoped tables, CREATE TABLE (not ALTER) for new tables, DROP in reverse dependency order.

Baseline today: event_details has is_sports_event (line ~791) but no is_kids_festival or kids_* tables.

3. Alter existing table

event_details — add column
ALTER TABLE public.event_details
    ADD COLUMN IF NOT EXISTS is_kids_festival boolean DEFAULT false NOT NULL;

COMMENT ON COLUMN public.event_details.is_kids_festival IS
    'When true, event uses Kids Festival flow (child profiles + competitions), not default ticket checkout.';

CREATE INDEX IF NOT EXISTS idx_event_details__is_kids_festival
    ON public.event_details(is_kids_festival)
    WHERE is_kids_festival = true;

For full schema file refresh: also add is_kids_festival bool DEFAULT false NOT NULL inside the CREATE TABLE public.event_details block (after is_sports_event).

4. New tables (CREATE)

4.1 kids_festival_settings

CREATE TABLE public.kids_festival_settings (
    id bigint DEFAULT nextval('public.sequence_generator'::regclass) NOT NULL,
    tenant_id character varying(255) NOT NULL,
    event_id bigint NOT NULL,
    registration_deadline timestamp without time zone,
    registration_open boolean DEFAULT true NOT NULL,
    allow_ticket_sales boolean DEFAULT false NOT NULL,
    points_first integer DEFAULT 5 NOT NULL,
    points_second integer DEFAULT 3 NOT NULL,
    points_third integer DEFAULT 1 NOT NULL,
    champion_enabled boolean DEFAULT true NOT NULL,
    champion_exclude_group_points boolean DEFAULT true NOT NULL,
    champion_max_category integer DEFAULT 3,
    results_display_mode character varying(32) DEFAULT 'FULL_NAME',
    eligibility_text text,
    created_at timestamp without time zone DEFAULT now() NOT NULL,
    updated_at timestamp without time zone DEFAULT now() NOT NULL,
    CONSTRAINT kids_festival_settings_pkey PRIMARY KEY (id),
    CONSTRAINT ux_kids_festival_settings__event UNIQUE (event_id),
    CONSTRAINT fk_kids_festival_settings__event
        FOREIGN KEY (event_id) REFERENCES public.event_details(id) ON DELETE CASCADE
);

4.2 kids_festival_day

CREATE TABLE public.kids_festival_day (
    id bigint DEFAULT nextval('public.sequence_generator'::regclass) NOT NULL,
    tenant_id character varying(255) NOT NULL,
    event_id bigint NOT NULL,
    day_label character varying(100) NOT NULL,
    event_date date NOT NULL,
    venue_name character varying(255) NOT NULL,
    venue_address character varying(500),
    sort_order integer DEFAULT 0 NOT NULL,
    notes text,
    created_at timestamp without time zone DEFAULT now() NOT NULL,
    updated_at timestamp without time zone DEFAULT now() NOT NULL,
    CONSTRAINT kids_festival_day_pkey PRIMARY KEY (id),
    CONSTRAINT fk_kids_festival_day__event
        FOREIGN KEY (event_id) REFERENCES public.event_details(id) ON DELETE CASCADE
);
CREATE INDEX idx_kids_festival_day__event_id ON public.kids_festival_day(event_id);

4.3 kids_competition

CREATE TABLE public.kids_competition (
    id bigint DEFAULT nextval('public.sequence_generator'::regclass) NOT NULL,
    tenant_id character varying(255) NOT NULL,
    event_id bigint NOT NULL,
    festival_day_id bigint,
    name character varying(255) NOT NULL,
    description text,
    competition_type character varying(20) NOT NULL,
    category_code character varying(20) NOT NULL,
    track character varying(50),
    fee_amount numeric(10,2) DEFAULT 0 NOT NULL,
    max_participants integer,
    min_group_size integer DEFAULT 3,
    max_group_size integer DEFAULT 10,
    time_limit_minutes integer,
    requires_soundtrack boolean DEFAULT false NOT NULL,
    judgment_criteria_json text,
    display_order integer DEFAULT 0 NOT NULL,
    is_active boolean DEFAULT true NOT NULL,
    created_at timestamp without time zone DEFAULT now() NOT NULL,
    updated_at timestamp without time zone DEFAULT now() NOT NULL,
    CONSTRAINT kids_competition_pkey PRIMARY KEY (id),
    CONSTRAINT chk_kids_competition__type
        CHECK (competition_type IN ('INDIVIDUAL', 'GROUP')),
    CONSTRAINT fk_kids_competition__event
        FOREIGN KEY (event_id) REFERENCES public.event_details(id) ON DELETE CASCADE,
    CONSTRAINT fk_kids_competition__day
        FOREIGN KEY (festival_day_id) REFERENCES public.kids_festival_day(id) ON DELETE SET NULL
);
CREATE INDEX idx_kids_competition__event_id ON public.kids_competition(event_id);

4.5 kids_competition_registration

CREATE TABLE public.kids_competition_registration (
    id bigint DEFAULT nextval('public.sequence_generator'::regclass) NOT NULL,
    tenant_id character varying(255) NOT NULL,
    event_id bigint NOT NULL,
    competition_id bigint NOT NULL,
    child_profile_id bigint NOT NULL,
    registration_status character varying(32) DEFAULT 'PENDING_PAYMENT' NOT NULL,
    fee_amount numeric(10,2) NOT NULL,
    effective_category character varying(20),
    stripe_payment_intent_id character varying(255),
    group_leader_registration_id bigint,
    registered_by_user_profile_id bigint NOT NULL,
    created_at timestamp without time zone DEFAULT now() NOT NULL,
    updated_at timestamp without time zone DEFAULT now() NOT NULL,
    CONSTRAINT kids_competition_registration_pkey PRIMARY KEY (id),
    CONSTRAINT ux_kids_reg__child_competition UNIQUE (competition_id, child_profile_id),
    CONSTRAINT fk_kids_reg__competition
        FOREIGN KEY (competition_id) REFERENCES public.kids_competition(id),
    CONSTRAINT fk_kids_reg__child
        FOREIGN KEY (child_profile_id) REFERENCES public.kids_child_profile(id),
    CONSTRAINT fk_kids_reg__event
        FOREIGN KEY (event_id) REFERENCES public.event_details(id) ON DELETE CASCADE
);
CREATE INDEX idx_kids_reg__event_id ON public.kids_competition_registration(event_id);

4.6 kids_competition_result

CREATE TABLE public.kids_competition_result (
    id bigint DEFAULT nextval('public.sequence_generator'::regclass) NOT NULL,
    tenant_id character varying(255) NOT NULL,
    event_id bigint NOT NULL,
    competition_id bigint NOT NULL,
    child_profile_id bigint,
    registration_id bigint,
    placement integer,
    points_awarded integer DEFAULT 0 NOT NULL,
    notes text,
    is_published boolean DEFAULT false NOT NULL,
    published_at timestamp without time zone,
    created_at timestamp without time zone DEFAULT now() NOT NULL,
    updated_at timestamp without time zone DEFAULT now() NOT NULL,
    CONSTRAINT kids_competition_result_pkey PRIMARY KEY (id),
    CONSTRAINT chk_kids_result__placement CHECK (placement IS NULL OR placement IN (1, 2, 3)),
    CONSTRAINT fk_kids_result__competition
        FOREIGN KEY (competition_id) REFERENCES public.kids_competition(id) ON DELETE CASCADE
);
CREATE INDEX idx_kids_result__event_published
    ON public.kids_competition_result(event_id, is_published);

6. Contrast with sports/arts results

Event Sports & Arts Competition Results attaches results to event_details for sports/arts-only events. Kids Festival uses kids_competition + kids_competition_result — no ticket or QR semantics. Do not reuse event_ticket_* for competition slots.
